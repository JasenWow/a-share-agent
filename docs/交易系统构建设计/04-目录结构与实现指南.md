## 2. 文件命名规范

### 2.1 通用规则

| 类型 | 规范 | 示例 |
|------|------|------|
| 目录名 | 全小写，连字符分隔 | `stock-screener/`, `factor-screen/` |
| Python 文件 | 全小写，下划线分隔 | `sync_agent_skills.py` |
| JSON 配置 | 全小写，连字符或下划线 | `plugin.json`, `screen.json` |
| Markdown | 全小写，连字符分隔 | `SKILL.md`, `architecture.md` |
| Parquet 文件 | 股票代码命名 | `000001.parquet`, `600519.parquet` |

### 2.2 Agent 插件目录结构

每个 Agent 插件遵循统一结构：

```
agent-plugins/<agent-name>/
├── AGENT.md              # Agent 清单（必需）
├── system-prompt.md      # 系统提示词（必需）
├── skills/               # 技能链接（可选，也可从 vertical plugin 引用）
│   └── <skill-name>.md   # 技能引用文件
├── commands/             # 命令定义（可选）
│   └── <cmd>.json
└── plugin.json           # 插件元数据（必需）
```

### 2.3 技能目录结构

```
vertical-plugins/a-share-analysis/skills/<skill-name>/
├── SKILL.md              # 技能说明文档（必需）
├── prompt.md             # 技能提示词模板（可选）
└── examples/             # 示例输入/输出（推荐）
    ├── input-example.md
    └── output-example.md
```

## 6. 同步与检查脚本

### 6.1 sync-agent-skills.py

```python
#!/usr/bin/env python3
"""
同步技能到 Agent 插件目录。
从 vertical-plugins 读取技能定义，更新引用到 agent-plugins。

用法:
  python scripts/sync-agent-skills.py [--check]
"""

import json
import shutil
from pathlib import Path

ROOT = Path(__file__).parent.parent
AGENT_PLUGINS = ROOT / "plugins" / "agent-plugins"
VERTICAL_PLUGINS = ROOT / "plugins" / "vertical-plugins"


def load_plugin_config(plugin_dir: Path) -> dict:
    """读取 plugin.json"""
    config_path = plugin_dir / "plugin.json"
    if not config_path.exists():
        return {}
    with open(config_path) as f:
        return json.load(f)


def resolve_skill(skill_ref: str) -> Path | None:
    """
    解析技能引用。
    格式: "a-share-analysis:factor-screen"
    → vertical-plugins/a-share-analysis/skills/factor-screen/
    """
    parts = skill_ref.split(":")
    if len(parts) != 2:
        return None
    plugin_name, skill_name = parts
    skill_path = VERTICAL_PLUGINS / plugin_name / "skills" / skill_name
    if skill_path.exists():
        return skill_path
    return None


def sync_agent(agent_dir: Path, dry_run: bool = False) -> list[str]:
    """同步单个 Agent 的技能引用"""
    config = load_plugin_config(agent_dir)
    if not config:
        return [f"SKIP {agent_dir.name}: no plugin.json"]

    results = []
    skills_dir = agent_dir / "skills"
    skills_dir.mkdir(exist_ok=True)

    # 清除旧引用
    if not dry_run:
        for f in skills_dir.iterdir():
            if f.is_file() and f.suffix == ".md":
                f.unlink()

    # 同步技能
    for skill_ref in config.get("skills", []):
        skill_path = resolve_skill(skill_ref)
        if not skill_path:
            results.append(f"WARN: skill not found: {skill_ref}")
            continue

        skill_name = skill_ref.split(":")[1]
        skill_md = skill_path / "SKILL.md"
        if not skill_md.exists():
            results.append(f"WARN: SKILL.md missing: {skill_ref}")
            continue

        target = skills_dir / f"{skill_name}.md"
        if not dry_run:
            target.write_text(
                f"<!-- Auto-synced from {skill_ref} -->\n"
                f"<!-- Source: {skill_path} -->\n\n"
                f"{skill_md.read_text()}"
            )
        results.append(f"OK: {skill_ref} -> {target.relative_to(agent_dir)}")

    return results


def main():
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument("--check", action="store_true", help="只检查，不同步")
    args = parser.parse_args()

    print(f"A-Share Agents Skill Sync {'(dry run)' if args.check else ''}")
    print("=" * 50)

    for agent_dir in sorted(AGENT_PLUGINS.iterdir()):
        if not agent_dir.is_dir():
            continue
        results = sync_agent(agent_dir, dry_run=args.check)
        for r in results:
            print(f"  [{agent_dir.name}] {r}")

    print("\nDone.")


if __name__ == "__main__":
    main()
```

### 6.2 check.py

```python
#!/usr/bin/env python3
"""
环境检查脚本 — 验证所有依赖和配置就绪。

用法:
  python scripts/check.py
"""

import json
import sys
from pathlib import Path

ROOT = Path(__file__).parent.parent


def check_python_packages() -> list[str]:
    """检查 Python 包依赖"""
    issues = []
    required = ["akshare", "tushare", "pandas", "openpyxl", "mcp", "fastapi"]
    for pkg in required:
        try:
            __import__(pkg)
        except ImportError:
            issues.append(f"MISSING: {pkg} (pip install {pkg})")
    return issues


def check_mcp_servers() -> list[str]:
    """检查 MCP 配置文件"""
    issues = []
    mcp_config = ROOT / ".mcp.json"
    if not mcp_config.exists():
        issues.append("MISSING: .mcp.json at project root")
        return issues

    with open(mcp_config) as f:
        config = json.load(f)

    required_servers = ["akshare", "tushare", "internal-store"]
    servers = config.get("mcpServers", {})
    for srv in required_servers:
        if srv not in servers:
            issues.append(f"MISSING: MCP server '{srv}' in .mcp.json")
        elif servers[srv].get("type") != "http":
            issues.append(f"WARN: MCP server '{srv}' type should be 'http'")

    return issues


def check_plugin_structure() -> list[str]:
    """检查插件目录结构"""
    issues = []

    # 检查 vertical plugin
    vp = ROOT / "plugins" / "vertical-plugins" / "a-share-analysis"
    if not vp.exists():
        issues.append(f"MISSING: {vp}")
    else:
        for required in ["plugin.json", ".mcp.json", "skills", "commands"]:
            if not (vp / required).exists():
                issues.append(f"MISSING: {vp / required}")

    # 检查 agent plugins
    agents = [
        "stock-screener", "equity-researcher", "factor-analyst",
        "backtester", "portfolio-manager", "market-monitor",
    ]
    ap = ROOT / "plugins" / "agent-plugins"
    for agent in agents:
        agent_dir = ap / agent
        if not agent_dir.exists():
            issues.append(f"MISSING: agent plugin {agent_dir}")
            continue
        for required in ["AGENT.md", "system-prompt.md", "plugin.json"]:
            if not (agent_dir / required).exists():
                issues.append(f"MISSING: {agent_dir / required}")

    return issues


def check_data_dir() -> list[str]:
    """检查数据目录"""
    issues = []
    data_dir = ROOT / "data"
    if not data_dir.exists():
        issues.append(f"MISSING: {data_dir} (will be auto-created on first run)")
    return issues


def check_env_vars() -> list[str]:
    """检查环境变量"""
    import os
    issues = []
    token = os.environ.get("TUSHARE_TOKEN", "")
    if not token:
        issues.append("WARN: TUSHARE_TOKEN not set (optional for AKShare-only mode)")
    return issues


def main():
    print("A-Share Agents Environment Check")
    print("=" * 50)

    all_issues = []

    checks = [
        ("Python Packages", check_python_packages),
        ("MCP Configuration", check_mcp_servers),
        ("Plugin Structure", check_plugin_structure),
        ("Data Directory", check_data_dir),
        ("Environment Variables", check_env_vars),
    ]

    for name, check_fn in checks:
        print(f"\n[{name}]")
        issues = check_fn()
        all_issues.extend(issues)
        if issues:
            for issue in issues:
                print(f"  - {issue}")
        else:
            print("  OK")

    print(f"\n{'=' * 50}")
    if all_issues:
        print(f"Found {len(all_issues)} issue(s). Fix before proceeding.")
        sys.exit(1)
    else:
        print("All checks passed!")
        sys.exit(0)


if __name__ == "__main__":
    main()
```
